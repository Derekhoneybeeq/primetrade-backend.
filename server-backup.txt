
"use strict";

const express = require("express");
const cors = require("cors");
const crypto = require("crypto");

const app = express();
const PORT = process.env.PORT || 10000;

// Fixed to Binance Spot TESTNET.
// Never change this to the live exchange for this project.
const BINANCE_BASE = "https://testnet.binance.vision";

app.use(cors());
app.use(express.json({ limit: "10kb" }));

// Simple request limit to discourage accidental repeated orders.
const requestCounts = new Map();

app.use("/api", (req, res, next) => {
  const now = Date.now();
  const ip = req.ip || "unknown";
  let entry = requestCounts.get(ip);

  if (!entry || now - entry.start > 60000) {
    entry = { start: now, count: 0 };
  }

  entry.count++;
  requestCounts.set(ip, entry);

  if (entry.count > 60) {
    return res.status(429).json({
      error: "Too many requests. Please wait a minute."
    });
  }

  next();
});

function credentialsAvailable() {
  return Boolean(
    process.env.BINANCE_API_KEY &&
    process.env.BINANCE_API_SECRET
  );
}

function signedQuery(params) {
  const query = new URLSearchParams(params).toString();

  const signature = crypto
    .createHmac("sha256", process.env.BINANCE_API_SECRET)
    .update(query)
    .digest("hex");

  return query + "&signature=" + signature;
}

async function binanceRequest(path, method, params) {
  const query = signedQuery(params);

  const options = {
    method,
    headers: {
      "X-MBX-APIKEY": process.env.BINANCE_API_KEY
    },
    signal: AbortSignal.timeout(15000)
  };

  let url = BINANCE_BASE + path;

  if (method === "GET") {
    url += "?" + query;
  } else {
    options.headers["Content-Type"] =
      "application/x-www-form-urlencoded";
    options.body = query;
  }

  const response = await fetch(url, options);
  const data = await response.json();

  if (!response.ok) {
    const error = new Error(
      data.msg || "Binance Testnet request failed"
    );
    error.status = response.status;
    error.exchangeCode = data.code;
    throw error;
  }

  return data;
}

app.get("/", (req, res) => {
  res.json({
    app: "PrimeTrade Backend",
    mode: "BINANCE SPOT TESTNET ONLY",
    message: "Backend is running."
  });
});

app.get("/api/health", (req, res) => {
  res.json({
    status: "ok",
    mode: "testnet",
    credentialsConfigured: credentialsAvailable()
  });
});

// Read virtual test-account balances.
app.get("/api/account", async (req, res) => {
  if (!credentialsAvailable()) {
    return res.status(503).json({
      error: "Testnet credentials are not configured on the server."
    });
  }

  try {
    const account = await binanceRequest(
      "/api/v3/account",
      "GET",
      {
        timestamp: Date.now(),
        recvWindow: 5000
      }
    );

    const balances = (account.balances || [])
      .filter(item => Number(item.free) > 0 ||
                      Number(item.locked) > 0)
      .map(item => ({
        asset: item.asset,
        free: item.free,
        locked: item.locked
      }));

    res.json({
      mode: "testnet",
      canTrade: account.canTrade,
      balances
    });
  } catch (error) {
    console.error("Account request failed:", error.message);

    res.status(error.status || 502).json({
      error: error.message,
      exchangeCode: error.exchangeCode || null
    });
  }
});

// Submit a real order on TESTNET virtual funds only.
app.post("/api/order", async (req, res) => {
  if (!credentialsAvailable()) {
    return res.status(503).json({
      error: "Testnet credentials are not configured."
    });
  }

  const symbol = String(req.body.symbol || "").toUpperCase();
  const side = String(req.body.side || "").toUpperCase();
  const type = String(req.body.type || "").toUpperCase();
  const quantity = Number(req.body.quantity);
  const allowedSymbols = ["BTCUSDT", "ETHUSDT", "BNBUSDT"];

  if (!allowedSymbols.includes(symbol)) {
    return res.status(400).json({
      error: "Unsupported trading pair."
    });
  }

  if (!["BUY", "SELL"].includes(side)) {
    return res.status(400).json({
      error: "Side must be BUY or SELL."
    });
  }

  if (!["MARKET", "LIMIT"].includes(type)) {
    return res.status(400).json({
      error: "Order type must be MARKET or LIMIT."
    });
  }

  if (!Number.isFinite(quantity) || quantity <= 0) {
    return res.status(400).json({
      error: "Quantity must be greater than zero."
    });
  }

  const params = {
    symbol,
    side,
    type,
    quantity: String(quantity),
    timestamp: Date.now(),
    recvWindow: 5000,
    newOrderRespType: "FULL"
  };

  if (type === "LIMIT") {
    const price = Number(req.body.price);

    if (!Number.isFinite(price) || price <= 0) {
      return res.status(400).json({
        error: "A valid limit price is required."
      });
    }

    params.price = String(price);
    params.timeInForce = "GTC";
  }

  try {
    const order = await binanceRequest(
      "/api/v3/order",
      "POST",
      params
    );

    res.json({
      mode: "testnet",
      message: "Binance Spot Testnet accepted the order.",
      order
    });
  } catch (error) {
    console.error("Test order failed:", error.message);

    res.status(error.status || 502).json({
      error: error.message,
      exchangeCode: error.exchangeCode || null
    });
  }
});

app.listen(PORT, "0.0.0.0", () => {
  console.log("PrimeTrade Testnet backend listening on port " + PORT);
});
